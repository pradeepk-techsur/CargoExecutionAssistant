  readonly field_name: string;

  readonly entryValues: Record<string, string | null>; // the 14 fields, AS SUBMITTED — no session/specialist identity (FR-9.16)

  readonly recommended_action: string;
  readonly rationale: string;

export const PROVIDER_FAILURE_REASONS = [

**`server/src/ai/outputSchema.ts`** — validates a PARSED JSON value (the
adapter, plan 05-04, does the HTTP fetch and JSON.parse; this module only
validates shape) against FR-9.7/FR-9.8 using `zod` (already a dependency).
Build a `.strict()` top-level object schema so ANY additional property —
including `hts_code`, `duty_amount`, `tariff_rate`, a classification code, an
admissibility ruling, a penalty or a `risk_score` field — is rejected by the
closed schema alone (FR-9.8 is satisfied by strictness; do not also hand-list
the forbidden names, that would be a second, driftable mechanism):

```ts
const ProposedValueSchema = z.object({
  field_name: z.enum(ENTRY_FIELDS), // from @cargoexec/contract — membership in the 14-field set
  proposed_value: z.string().trim().min(1).max(2000),
  addresses_rule_ids: z.array(z.string().regex(/^RIV-[0-9]{3}$/)).min(1),
}).strict();

const ProviderResponseSchema = z.object({
  recommended_action: z.string().trim().min(1).max(500),
  rationale: z.string().trim().min(1).max(2000),
  proposed_values: z.array(ProposedValueSchema).min(0).max(14),
}).strict();
```

After the zod parse succeeds, apply the two checks zod cannot express
declaratively:
1. No duplicate `field_name` across `proposed_values` (FR-9.7 "duplicate
   field_name ⇒ invalid").
2. Every `addresses_rule_ids` entry MUST be a member of the rule ids actually
   present in `req.findings` for THIS exception (FR-9.7's "rule ids present in
   this exception's findings") — so `validateProviderResponse` takes the
   request's findings as a second argument to cross-check against, not just the
   raw response:
   ```ts
   export function validateProviderResponse(
     raw: unknown,
     knownRuleIds: ReadonlySet<string>,
   ): { ok: true; value: z.infer<typeof ProviderResponseSchema> } | { ok: false } {
     const parsed = ProviderResponseSchema.safeParse(raw);
     if (!parsed.success) return { ok: false };
     const seen = new Set<string>();
     for (const v of parsed.data.proposed_values) {
       if (seen.has(v.field_name)) return { ok: false };
       seen.add(v.field_name);
       for (const rid of v.addresses_rule_ids) {
         if (!knownRuleIds.has(rid)) return { ok: false };
       }
     }
     return { ok: true, value: parsed.data };
   }
   ```
Also reject rationale containing an internal rule id (`/RIV-[0-9]{3}/`) — FR-9.9
"no rule ids ... in the rationale" — as part of the same `ok:false` path (scan
`rationale` for the rule-id pattern after the zod parse).

**`server/src/ai/fakeProvider.ts`** — the no-network test double (TechArch §5):
```ts
export const FAKE_AI_TRIGGERS: Record<ProviderFailureReason, string> = {
  PROVIDER_TIMEOUT: '__FAKE_AI_TIMEOUT__',
  PROVIDER_UNAVAILABLE: '__FAKE_AI_UNAVAILABLE__',
  PROVIDER_RATE_LIMITED: '__FAKE_AI_RATE_LIMITED__',
  PROVIDER_AUTH_FAILED: '__FAKE_AI_AUTH_FAILED__',
  SCHEMA_INVALID: '__FAKE_AI_SCHEMA_INVALID__',
  CONTENT_FILTERED: '__FAKE_AI_CONTENT_FILTERED__',
  INTERNAL_ERROR: '__FAKE_AI_INTERNAL_ERROR__',
};
export function createFakeProvider(): RecommendationProvider {
  return {
    async generate(req) {
      const allValues = Object.values(req.entryValues);
      for (const [reason, marker] of Object.entries(FAKE_AI_TRIGGERS)) {
        if (allValues.some((v) => v === marker)) {
          return { outcome: 'FAILURE', failure_reason: reason as ProviderFailureReason };
        }
      }
      // Default: deterministic success addressing every finding.
      const byField = new Map<string, string[]>(); // field_name -> rule_ids
      for (const f of req.findings) {
        const arr = byField.get(f.field_name) ?? [];
        arr.push(f.rule_id);
        byField.set(f.field_name, arr);
      }
      const proposed_values = [...byField.entries()].map(([field_name, ruleIds]) => ({
        field_name,
        proposed_value: `Corrected value for ${field_name}`,
        addresses_rule_ids: ruleIds,
      }));
      return {
        outcome: 'SUCCESS',
        recommended_action: 'Review and correct the flagged fields, then resolve the case.',
        rationale:
          `This entry did not satisfy ${req.findings.length} required-information rule(s). ` +
          'The suggested values above would address each flagged field. Review them against ' +
          'the original documentation before deciding.',
        proposed_values,
      };
    },
  };
}
```
No `setTimeout`, no real delay, no network call anywhere in this file — it
resolves synchronously-fast so unit/db/api tests stay fast, and it is safe to
wire into the REAL running server (via `AI_PROVIDER_URL=fake:deterministic`)
for e2e demonstration with no external dependency (plan 05-06).

## Trust Boundaries

| Boundary | Description |
|----------|--------------|
| provider→server | The (eventual, plan 05-04) hosted LLM response is untrusted input; this plan's `outputSchema.ts` is the FIRST gate it passes through before any of its content can reach a database write or a screen. |
| env→server | `AI_API_KEY`/`AI_PROVIDER_URL` are operator-supplied configuration; `config.ts`'s self-checks are the trust boundary between "the environment" and "a running server". |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|------------------|
| T-05-01 | Tampering | `outputSchema.ts::validateProviderResponse` — a hostile/malformed provider response | mitigate | `.strict()` zod schema rejects any additional top-level property (closing off FR-9.8's excluded-determinations list structurally, not by denylist); duplicate `field_name` and out-of-set `addresses_rule_ids` are rejected by explicit post-parse checks in the same function, BEFORE the value ever reaches a caller (plan 05-04's job treats any `ok:false` as terminal `SCHEMA_INVALID`, never retried). |
| T-05-02 | Information disclosure | `config.ts` — `AI_API_KEY` | mitigate | Every `ConfigError` thrown by the new self-checks names the KEY only (`AI_API_KEY`, `AI_PROVIDER_URL`, …), never the value — matching the existing four self-checks' §4.7 convention; proven by `config.spec.ts`'s existing "names the key but discloses no neighbouring env value" pattern, extended to the new keys. |
| T-05-03 | Tampering | `promptManifest.ts` — the prompt template is the trust anchor for "what did the AI say" (traceability, FR-9.10) | mitigate | The digest self-check at boot (`computeFileDigestSync` vs `manifest.json`) refuses to start the server if the shipped template text was edited without a version bump, so a `prompt_version` recorded on a recommendation always corresponds to the exact template text that produced it. |
