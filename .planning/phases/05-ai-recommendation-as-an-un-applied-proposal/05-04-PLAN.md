```ts
export function createHttpProvider(opts: {
  url: string; apiKey: string; modelId: string; timeoutMs: number;
}): RecommendationProvider {
  return {
    async generate(req) {
      const body = renderPromptRequest(req); // { model: opts.modelId, messages/prompt: ..., response_format: 'json' } — an OpenAI-compatible chat/completions shape is a reasonable default; the exact wire shape is this adapter's own concern, invisible to every caller
      const totalBudgetMs = 45_000;
      const start = Date.now();
      let lastErr: { failure_reason: ProviderFailureReason } | undefined;
      for (let attempt = 0; attempt < 2; attempt++) {
        if (attempt === 1) {
          await sleep(2000); // the ONE retry, after a 2s delay (FR-9.5)
        }
        if (Date.now() - start > totalBudgetMs) break;
        const controller = new AbortController();
        const timer = setTimeout(() => controller.abort(), opts.timeoutMs);
        try {
          const res = await fetch(opts.url, {
            method: 'POST',
            headers: {
              'content-type': 'application/json',
              authorization: `Bearer ${opts.apiKey}`, // NEVER logged
            },
            body: JSON.stringify(body),
            signal: controller.signal,
          });
          clearTimeout(timer);
          if (res.status === 401 || res.status === 403) {
            return { outcome: 'FAILURE', failure_reason: 'PROVIDER_AUTH_FAILED' };
          }
          if (res.status === 429) {
            lastErr = { failure_reason: 'PROVIDER_RATE_LIMITED' };
            continue; // retryable
          }
          if (res.status >= 500) {
            lastErr = { failure_reason: 'PROVIDER_UNAVAILABLE' };
            continue; // retryable
          }
          if (!res.ok) {
            return { outcome: 'FAILURE', failure_reason: 'INTERNAL_ERROR' };
          }
          let json: unknown;
          try {
            json = await res.json();
          } catch {
            return { outcome: 'FAILURE', failure_reason: 'SCHEMA_INVALID' }; // parse failure is terminal (FR-9.6), never retried
          }
          const knownRuleIds = new Set(req.findings.map((f) => f.rule_id));
          const validated = validateProviderResponse(extractPayload(json), knownRuleIds);
          if (!validated.ok) {
            return { outcome: 'FAILURE', failure_reason: 'SCHEMA_INVALID' };
          }
          return { outcome: 'SUCCESS', ...validated.value };
        } catch (err) {
          clearTimeout(timer);
          if ((err as { name?: string }).name === 'AbortError') {
            lastErr = { failure_reason: 'PROVIDER_TIMEOUT' };
            continue; // retryable
          }
          lastErr = { failure_reason: 'PROVIDER_UNAVAILABLE' }; // connection failure
          continue;
        }
      }
      return { outcome: 'FAILURE', failure_reason: lastErr?.failure_reason ?? 'PROVIDER_UNAVAILABLE' };
    },
  };
}
```

**`server/src/ai/job.ts`** — the per-exception generation job (F9 §Process
steps 3–9). Reads over `aiPool` (`cargoexec_ai`); the terminal write runs over
`appPool` (`cargoexec_app`) inside ONE `withTransaction` — see plan 05-02's
plan-body for exactly why this pool split is required (`append()`'s
case-anchor lock needs `UPDATE` on `cargo_entries`, granted only to
`cargoexec_app`).

```ts
import type { Pool } from 'pg';
import { withTransaction } from '../db/tx.js';
import { append } from '../services/audit/writer.js';
import { loadExceptionForGeneration } from '../db/repositories/exceptions.js';
import { loadEntryValuesForRecommendation } from '../db/repositories/entries.js';
import { loadValidationByEntry } from '../db/repositories/validation.js';
import { loadRecommendationByException } from '../db/repositories/recommendations.js';
import {
  markRecommendationAvailable, markRecommendationUnavailable, insertRecommendationValues,
} from '../db/repositories/recommendations.js';
import type { RecommendationProvider } from './provider.js';
import { logger } from '../http/logger.js';

export async function runGenerationJob(
  deps: { aiPool: Pool; appPool: Pool; provider: RecommendationProvider; modelId: string; promptVersion: string },
  exceptionId: string,
): Promise<void> {
  const start = Date.now();
  try {
    // 1. Cheap pre-check over the AI pool (step 3, FR-9.17): skip the provider
    //    call entirely if this exception's recommendation is already resolved.
    //    This is an OPTIMISATION, not the authoritative guard — the conditional
    //    UPDATE below is the real idempotence mechanism.
    const current = await loadRecommendationByException(deps.aiPool, exceptionId);
    if (current !== null && current.status !== 'PENDING') return;

    const ex = await loadExceptionForGeneration(deps.aiPool, exceptionId);
    if (ex === null) {
      logger.warn({ exception_id: exceptionId }, 'generation job: exception not found, exiting');
      return;
    }
    const [entryValues, validation] = await Promise.all([
      loadEntryValuesForRecommendation(deps.aiPool, ex.entry_id),
      loadValidationByEntry(deps.aiPool, ex.entry_id),
    ]);
    if (entryValues === null || validation === null) {
      logger.warn({ exception_id: exceptionId }, 'generation job: entry/validation vanished, exiting');
      return;
    }

    const result = await deps.provider.generate({
      entryValues,
      findings: validation.findings,
      ruleSetVersion: validation.rule_set_version,
      promptVersion: deps.promptVersion,
    });
    const latencyMs = Date.now() - start;

    if (current === null) return; // recommendation row must exist (F5 FR-5.12); defensive.

    if (result.outcome === 'SUCCESS') {
      await withTransaction(deps.appPool, async (tx) => {
        const { updated } = await markRecommendationAvailable(tx, {
          recommendation_id: current.id,
          recommended_action: result.recommended_action,
          rationale: result.rationale,
          model_id: deps.modelId,
          prompt_version: deps.promptVersion,
          latency_ms: latencyMs,
        });
        if (!updated) return; // already resolved by a concurrent/duplicate dispatch (FR-9.17)
        await insertRecommendationValues(tx, current.id, result.proposed_values);
        await append(tx, {
          case_id: ex.entry_id,
          exception_id: exceptionId,
          recommendation_id: current.id,
          action_type: 'RECOMMENDATION_GENERATED',
          actor: { type: 'AI' },
          before_state: 'PENDING',
          after_state: 'AVAILABLE',
          values: result.proposed_values.map((v) => ({
            field_name: v.field_name,
            before_value: (entryValues as Record<string, string | null>)[v.field_name] ?? null,
            before_origin: (entryValues as Record<string, string | null>)[v.field_name] != null ? 'HUMAN' : null,
            after_value: v.proposed_value,
            after_origin: 'AI',
          })),
        });
      });
    } else {
      await withTransaction(deps.appPool, async (tx) => {
        const { updated } = await markRecommendationUnavailable(tx, {
          recommendation_id: current.id,
          failure_reason: result.failure_reason,
          model_id: deps.modelId,
          prompt_version: deps.promptVersion,
        });
        if (!updated) return;
        await append(tx, {
          case_id: ex.entry_id,
          exception_id: exceptionId,
          recommendation_id: current.id,
          action_type: 'RECOMMENDATION_UNAVAILABLE',
          actor: { type: 'AI' },
          before_state: 'PENDING',
          after_state: 'UNAVAILABLE',
        });
      });
    }
  } catch (err) {
    // A truly unexpected failure (a provider.generate() that THROWS rather
    // than resolving a FAILURE outcome, a transient DB error) must not crash
    // the worker or leave the recommendation silently PENDING forever without
    // a trace. Log loudly; the recommendation simply stays PENDING (F9's
    // §Error States: no sweeper/retry in v1 — FR-9.19, presented pending-then-
    // stale by F10). Do NOT attempt a write here: we may not know which pool
    // is safe to use, and a half-understood failure must not become a
    // fabricated UNAVAILABLE outcome.
    logger.error({ exception_id: exceptionId, err }, 'generation job failed unexpectedly; recommendation remains PENDING');
  }
}
```

**`server/src/ai/worker.ts`** — the bounded-concurrency in-process dispatcher
(TechArch §5, no scheduler/cron/queue library — PRD §10 excludes
`bullmq`/`bull`/`agenda`/`node-cron` etc., already enforced by

```ts
export function createRecommendationWorker(deps: {
  concurrency: number;
  runJob: (exceptionId: string) => Promise<void>;
}): { dispatch(exceptionId: string): void } {
  const queue: string[] = [];
  let active = 0;

  function pump(): void {
    while (active < deps.concurrency && queue.length > 0) {
      const id = queue.shift() as string;
      active += 1;
      void deps
        .runJob(id)
        .catch((err) => {
          // runJob already catches internally; this is belt-and-braces so a
          // programming error here can never take down the process.
          logger.error({ exception_id: id, err }, 'recommendation worker: unexpected job rejection');
        })
        .finally(() => {
          active -= 1;
          pump();
        });
    }
  }

  return {
    dispatch(exceptionId: string): void {
      queue.push(exceptionId);
      pump();
    },
  };
}
```
